"""One-use successor trial ordering. No CLI, transport, or claim registration.

The concrete driver must authenticate every observation and bind its code,
artifacts and subprocess receipts before this engine may be admitted live.
Successful recovery never converts a failed target trial to PASS.
"""
import hashlib
import json
import os
from pathlib import Path
import re
import time

ROOT = Path(__file__).resolve().parent
OUTPUT = ROOT / 'execution'
SOURCE = dict(bundle='<REDACTED_SOURCE_1>', release='<REDACTED_SOURCE_2>')
TARGET = dict(bundle='<REDACTED_TARGET_1>', release='<REDACTED_TARGET_2>')
FALLBACK = dict(bundle='<REDACTED_FALLBACK_1>', release='<REDACTED_FALLBACK_2>')
OLD = '<REDACTED_OLD_1>'
PENDING = '<REDACTED_PENDING_1>'
HEALTHY = '<REDACTED_HEALTHY_1>'
PHASES = ('preflight', 'arm_state', 'install_exitrd', 'request_source_reboot',
          'await_fastboot', 'capture_start', 'boot_once', 'observe_target',
          'close_capture', 'post_capture_health', 'inspect_source_abort',
          'restore_source_exitrd', 'restore_source_state', 'verify_source_abort',
          'locate_fallback', 'recovery_capture_start', 'reboot_installed',
          'observe_fallback', 'close_recovery_capture', 'stage_fallback', 'restore_fallback_state',
          'verify_fallback_restoration')
MUTATIONS = frozenset({'arm_state', 'install_exitrd', 'request_source_reboot',
                       'boot_once', 'restore_source_exitrd', 'restore_source_state',
                       'reboot_installed', 'stage_fallback', 'restore_fallback_state'})


def need(ok, message):
    if not ok:
        raise ValueError(message)


def boot_id(value):
    need(type(value) is str and re.fullmatch(r'[0-9a-f]{8}(?:-[0-9a-f]{4}){3}-[0-9a-f]{12}', value)
         and value != '<BOOT_ID_1>', 'invalid boot identity')


def same_identity(value, expected, *, excluded=()):
    need(type(value) is dict and set(value) == {'boot_id', 'bundle', 'release'}, 'identity fields')
    boot_id(value['boot_id'])
    need(all(value.get(key) == item for key, item in expected.items())
         and value['boot_id'] not in excluded, 'wrong or reused boot identity')


def save(path, value):
    raw = (json.dumps(value, sort_keys=True, indent=2, allow_nan=False) + '\n').encode()
    need(len(raw) <= 1048576, 'receipt bound')
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW | os.O_CLOEXEC, 0o600)
    try:
        view = memoryview(raw)
        while view:
            count = os.write(fd, view)
            need(count > 0, 'receipt write stalled')
            view = view[count:]
        os.fsync(fd)
    finally:
        os.close(fd)
    parent = os.open(path.parent, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW | os.O_CLOEXEC)
    try:
        os.fsync(parent)
    finally:
        os.close(parent)
    return hashlib.sha256(raw).hexdigest()


class Controller:
    def __init__(self, driver, context):
        need(set(context) == {'source', 'owner', 'admission_sha256'}, 'controller context fields')
        same_identity(context['source'], SOURCE)
        need(re.fullmatch('[0-9a-f]{32}', context['owner'])
             and re.fullmatch('[0-9a-f]{64}', context['admission_sha256'])
             and context['admission_sha256'] != '0' * 64, 'owner/admission binding')
        need(tuple(driver.PHASES) == PHASES, 'incomplete concrete driver')
        self.context = json.loads(json.dumps(context))
        self.driver = driver
        self.entered = []
        self.results = {}
        self.errors = []
        self.target_identity = None
        self.selection_restored = False
        self.output = OUTPUT
        need(self.output == ROOT / 'execution' and ROOT.resolve() == ROOT, 'fixed canonical execution path')
        self.output.mkdir(mode=0o700)  # Any prior attempt refuses before callbacks.
        save(self.output / 'context.json', self.context)

    def action(self, phase):
        need(phase in PHASES and phase not in self.entered, 'unknown or repeated phase')
        try:
            # Closing an owned receiver must work even after source/USB changes.
            # Its own method authenticates ownership, original deadline and cleanup.
            if phase not in ('close_capture', 'close_recovery_capture'):
                self.driver.gate(phase, self.context, self.results)
            self.entered.append(phase)
            intent = dict(phase=phase, mutation=phase in MUTATIONS,
                          admission_sha256=self.context['admission_sha256'],
                          owner=self.context['owner'], monotonic=time.monotonic(), prior=list(self.results))
            pin = save(self.output / (phase + '-entered.json'), intent)
            value = getattr(self.driver, phase)(self.context, self.results, pin)
            need(type(value) is dict and value.get('status') == 'PASS', 'phase did not pass: ' + phase)
            save(self.output / (phase + '-result.json'), value)
            self.results[phase] = value
            return value
        except Exception as error:
            failure = dict(phase=phase, reason=str(error)[:1000], type=type(error).__name__, retry_permitted=False)
            self.errors.append(failure)
            save(self.output / (phase + '-failure.json'), failure)
            raise

    def require_state(self, value, identity, before, after):
        same_identity(value.get('identity'), identity)
        need(value.get('before_sha256') in before and value.get('after_sha256') == after
             and value.get('completed') is True, 'incomplete state operation proof')

    def target_health(self, value):
        identity = value.get('identity')
        same_identity(identity, TARGET, excluded=(self.context['source']['boot_id'],))
        need(value.get('state_sha256') == HEALTHY and value.get('current_boot_healthy') is True
             and value.get('readiness_boot_bound') is True and value.get('physical_guards_passed') is True,
             'target health is absent, stale or incomplete')
        if self.target_identity is not None:
            need(identity == self.target_identity, 'target changed during full capture')
        self.target_identity = dict(identity)

    def close_capture(self):
        value = self.action('close_capture')
        need(value.get('cleanup_complete') is True, 'capture cleanup incomplete')
        if value.get('capture_started') is True:
            need(value.get('full_lifetime') is True, 'capture ended before original deadline')
        else:
            need(value.get('capture_started') is False and value.get('no_owned_process') is True
                 and 'boot_once' not in self.entered, 'capture absence is unproven')
        return value

    def abort_source(self):
        # Never infer reboot absence from a failed SSH request or missing reply.
        need('request_source_reboot' not in self.entered, 'reboot may have been requested')
        value = self.action('inspect_source_abort')
        same_identity(value.get('identity'), self.context['source'])
        need(value.get('reboot_intent_absent') is True and value.get('physical_guards_passed') is True
             and value.get('shutdown') in ('original', 'patched')
             and value.get('state_sha256') in (OLD, PENDING), 'source abort is not proven safe')
        if value['shutdown'] == 'patched':
            restored = self.action('restore_source_exitrd')
            need(restored.get('shutdown_restored') is True and restored.get('reboot_requested') is False,
                 'source exitrd not restored')
            same_identity(restored.get('identity'), self.context['source'])
        if value['state_sha256'] == PENDING:
            restored = self.action('restore_source_state')
            self.require_state(restored, self.context['source'], (PENDING,), OLD)
        proof = self.action('verify_source_abort')
        same_identity(proof.get('identity'), self.context['source'])
        need(proof.get('state_sha256') == OLD and proof.get('shutdown_restored') is True
             and proof.get('physical_guards_passed') is True, 'source abort verification incomplete')
        self.selection_restored = True

    def restore_fallback(self):
        # No dependency on target stage/switch-root: an early failure can return
        # straight to V11. Only fresh authenticated physical evidence permits it.
        value = self.action('locate_fallback')
        if value.get('mode') == 'fastboot':
            need(value.get('device_matches') is True and value.get('installed_fallback_route_verified') is True,
                 'installed fallback recovery route is unproven')
            # Full target capture has already closed. A separately owned full
            # fallback capture precedes one ordinary reboot of the installed
            # signed selector. No second RAM boot or flashing route exists.
            captured = False
            try:
                ready = self.action('recovery_capture_start')
                need(ready.get('live') is True and ready.get('remaining_lifetime_valid') is True,
                     'fallback receiver not ready')
                self.action('reboot_installed')
                value = self.action('observe_fallback')
            finally:
                if 'recovery_capture_start' in self.entered:
                    closed = self.action('close_recovery_capture')
                    captured = (closed.get('capture_started') is True and closed.get('full_lifetime') is True
                                and closed.get('cleanup_complete') is True and closed.get('capture_status') == 'PASS')
                    need(captured, 'fallback full capture/cleanup incomplete')
            need(captured, 'fallback capture did not close')
        excluded = [self.context['source']['boot_id']]
        if self.target_identity:
            excluded.append(self.target_identity['boot_id'])
        identity = value.get('identity')
        same_identity(identity, FALLBACK, excluded=excluded)
        need(value.get('authenticated') is True and value.get('physical_guards_passed') is True
             and value.get('state_sha256') in (PENDING, HEALTHY), 'fallback is not authenticated')
        staged = self.action('stage_fallback')
        same_identity(staged.get('identity'), identity)
        need(staged.get('ram_files_verified') is True, 'fallback RAM staging incomplete')
        restored = self.action('restore_fallback_state')
        self.require_state(restored, identity, (value['state_sha256'],), OLD)
        proof = self.action('verify_fallback_restoration')
        same_identity(proof.get('identity'), identity)
        need(proof.get('state_sha256') == OLD and proof.get('physical_guards_passed') is True
             and proof.get('selection_eligibility_restored') is True,
             'fallback restoration verification incomplete')
        self.selection_restored = True

    def run(self):
        trial_passed = False
        capture_closed = False
        try:
            preflight = self.action('preflight')
            same_identity(preflight.get('identity'), self.context['source'])
            need(preflight.get('ram_staging_verified') is True, 'source RAM staging absent')
            armed = self.action('arm_state')
            self.require_state(armed, self.context['source'], (OLD,), PENDING)
            installed = self.action('install_exitrd')
            same_identity(installed.get('identity'), self.context['source'])
            need(installed.get('installed') is True and installed.get('reboot_requested') is False,
                 'exitrd installation incomplete')
            self.action('request_source_reboot')
            self.action('await_fastboot')
            capture = self.action('capture_start')
            need(capture.get('live') is True and capture.get('remaining_lifetime_valid') is True,
                 'full receiver not ready')
            self.action('boot_once')
            self.target_health(self.action('observe_target'))
        except Exception as error:
            save(self.output / 'trial-failure.json', dict(reason=str(error)[:1000], retry_permitted=False))
            if not self.errors:
                self.errors.append(dict(phase='trial-validation', reason=str(error)[:1000]))
        finally:
            if 'capture_start' in self.entered:
                try:
                    closed = self.close_capture()
                    capture_closed = True
                    if closed.get('capture_status') != 'PASS':
                        self.errors.append(dict(phase='capture-result', reason='full target capture did not pass'))
                except Exception as error:
                    if not any(row['phase'] == 'close_capture' for row in self.errors):
                        self.errors.append(dict(phase='capture-closure', reason=str(error)[:1000]))
        if not self.errors and self.target_identity and capture_closed:
            try:
                self.target_health(self.action('post_capture_health'))
                trial_passed = True
            except Exception as error:
                if not self.errors:
                    self.errors.append(dict(phase='post-capture-validation', reason=str(error)[:1000]))
        if not trial_passed and 'arm_state' in self.entered:
            try:
                if 'request_source_reboot' not in self.entered:
                    self.abort_source()
                elif 'capture_start' not in self.entered or capture_closed:
                    self.restore_fallback()
            except Exception as error:
                save(self.output / 'restoration-failure.json', dict(reason=str(error)[:1000], retry_permitted=False))
                if not self.errors:
                    self.errors.append(dict(phase='restoration', reason=str(error)[:1000]))
        result = dict(status='COMPONENT_PASS' if trial_passed else 'FAIL',
                      target_identity=self.target_identity, successor_running=trial_passed,
                      selection_eligibility_restored=self.selection_restored,
                      phases=list(self.results), entered=self.entered, errors=self.errors,
                      release_qualified=False, retry_permitted=False)
        save(self.output / 'result.json', result)
        return result


if __name__ == '__main__':
    raise SystemExit('No live entrypoint: exact concrete driver/admission must be qualified first')
