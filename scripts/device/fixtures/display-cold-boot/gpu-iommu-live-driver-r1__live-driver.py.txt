"""Assemble exact controller components; full admission and root bridge required.

No CLI, default allow gate, claim registration or automatic phone execution.
"""
import hashlib
import importlib.util
import json
import os
from pathlib import Path
import stat

HERE = Path(__file__).resolve().parent
OUTPUT = HERE.parent / 'gpu-iommu-controller-r1/execution'
PATHS = {'route-checks.py': Path('/fixture/private-dependency-0'), 'boot-callbacks.py': Path('/fixture/private-dependency-1'), 'capture-callbacks.py': Path('/fixture/private-dependency-2'), 'fastboot-callbacks.py': Path('/fixture/private-dependency-3')}
PINS = {'route-checks.py': '9eaa453c108d8c16cad504c32f75f1278a8bab0578823af6cc4c3c16b63bd761', 'boot-callbacks.py': 'bf7413425b7581a8bfcd064786c4316ded93b85f445713ee63126ff27966d38e', 'capture-callbacks.py': '139e567d978552fceba771bfe66480cce5334c04f50f4e98fd4d0c0d9eda202b', 'fastboot-callbacks.py': '22de83b70a9e9f62c41d228f104a3e53a54daea3d08ea944640113d43df55932'}


def pins():
    for name, digest in PINS.items():
        path = PATHS[name]
        if path.resolve() != path or path.stat().st_size > 262144 or hashlib.sha256(path.read_bytes()).hexdigest() != digest:
            raise ValueError('assembled driver producer changed')


def load(name, filename):
    spec = importlib.util.spec_from_file_location(name, PATHS[filename])
    m = importlib.util.module_from_spec(spec); spec.loader.exec_module(m); return m


pins()
ROUTE = load('driver_route', 'route-checks.py')
BOOT = load('driver_boot', 'boot-callbacks.py')
CAPTURE = load('driver_capture', 'capture-callbacks.py')
FASTBOOT = load('driver_fastboot', 'fastboot-callbacks.py')
R = ROUTE.R
A, C, need = R.A, R.A.C, R.need
HEALTH = FASTBOOT.HEALTH


class Driver:
    PHASES = C.PHASES

    def __init__(self, source, before_phase, authorize, capture_bridge, transport=None):
        pins()
        need(type(source) is dict and source.get('clean') is True
             and callable(before_phase) and callable(authorize), 'complete driver admission required')
        need(all(callable(getattr(capture_bridge, name, None)) for name in ('command', 'launch', 'stop')),
             'complete capture bridge required before source mutation')
        self.source = A.decode(json.dumps(source))
        self.before_phase, self.authorize = before_phase, authorize
        self.route = ROUTE.Route(self.source, authorize)
        self.actions = A.Actions(self.source, authorize, transport=transport)
        self.source_reads = R.SourceReads(self.source, authorize)
        self.captures = CAPTURE.Captures(self.source, authorize, capture_bridge)
        self.health = HEALTH.HealthReads(self.source, authorize, transport=transport)
        self.fastboot = FASTBOOT.Fastboot(self.source, authorize, self.route.check, self.fallback_capture, transport=transport)
        self.boot = BOOT.Boot(self.source, authorize, self.route.check, self.target_capture)
        self.bindings = {'preflight': self.route.prepare}
        for provider, phases in (
            (self.actions, A.PHASES),
            (self.source_reads, ('inspect_source_abort', 'verify_source_abort')),
            (self.health, ('observe_target', 'post_capture_health', 'observe_fallback', 'verify_fallback_restoration', 'verify_target_restoration')),
            (self.fastboot, FASTBOOT.PHASES),
            (self.captures, CAPTURE.PHASES),
            (self.boot, BOOT.PHASES),
        ):
            for phase in phases:
                need(phase not in self.bindings, 'duplicate controller phase binding')
                self.bindings[phase] = getattr(provider, phase)
        need(set(self.bindings) == set(C.PHASES) and all(callable(f) for f in self.bindings.values()),
             'incomplete concrete controller phase map')
        self.output_paths()

    def output_paths(self):
        # Check every concrete writer. Receipt-only adapters take explicit paths;
        # they do not own an output directory. Capture's writer is its controller.
        need(all(path == OUTPUT for path in (
            C.OUTPUT, A.OUTPUT, R.OUTPUT, ROUTE.OUTPUT, BOOT.OUTPUT,
            CAPTURE.OUTPUT, FASTBOOT.OUTPUT, HEALTH.OUTPUT,
            BOOT.A.OUTPUT, BOOT.A.C.OUTPUT, CAPTURE.A.C.OUTPUT,
            FASTBOOT.A.OUTPUT, FASTBOOT.A.C.OUTPUT, HEALTH.A.OUTPUT, HEALTH.A.C.OUTPUT)),
            'controller component output paths differ')

    def target_capture(self, context, results):
        return self.captures.ready('target', context, results)

    def fallback_capture(self, context, results):
        return self.captures.ready('fallback', context, results)

    def gate(self, phase, context, results):
        need(phase in self.PHASES and phase not in ('close_capture', 'close_recovery_capture'),
             'driver phase gate is not a cleanup gate')
        pins(); self.output_paths()
        need(os.geteuid() == 1000 and OUTPUT.resolve() == OUTPUT and OUTPUT.is_dir()
             and OUTPUT.stat().st_uid == 1000 and stat.S_IMODE(OUTPUT.stat().st_mode) == 0o700,
             'driver execution directory owner')
        need(A.exact_json(context['source'], dict(C.SOURCE, boot_id=A.SOURCE_BOOT))
             and context['owner'] == A.OWNER, 'driver source/custody changed')
        need(A.exact_json(A.decode(R.receipt(OUTPUT / 'context.json', 4096)), context), 'driver context changed')
        need(A.exact_json(ROUTE.W.source_identity(), self.source), 'driver host source changed')
        for name, value in results.items():
            need(name in self.PHASES and A.exact_json(
                A.decode(R.receipt(OUTPUT / (name + '-result.json'), 1048576)), value), 'driver prior result changed')
        need(self.before_phase(phase, context, results) is True, 'driver phase admission refused')

    def __getattr__(self, phase):
        bindings = self.__dict__.get('bindings', {})
        if phase not in bindings:
            raise AttributeError(phase)
        return bindings[phase]


if __name__ == '__main__':
    raise SystemExit('Import-only assembled driver; complete admission/root bridge required')
