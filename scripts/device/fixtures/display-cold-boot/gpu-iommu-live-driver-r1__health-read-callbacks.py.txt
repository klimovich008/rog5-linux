"""Authenticated health callbacks; complete outer admission remains required.

Initial observations poll only read-only identity/readiness for at most 300 s.
Health and restoration probes run once. No capture deadline is shortened here.
The pinned worker currently needs an existing route when invoked as deck.
"""
import base64
import importlib.util
import json
import math
import os
from pathlib import Path
import stat
import time

HERE = Path(__file__).resolve().parent
OUTPUT = HERE.parent / 'gpu-iommu-controller-r1/execution'
PINS = {
    'receipt-common.py': '50e21650fb00b1babc6eaee9916246688afc401209392483f728666fdc4ddfd4',
    'successor-health.py': '738f5d0b6bc6aef9f7b46babd7f53a46103696e8ce9624c21f2454d3de9bcb32',
    'fallback-observation.py': 'db1f2828fff39d0e7fa1aaa679fef6e44876a00873b1bc235cc3f598fabc4c10',
}
PATHS = {name: HERE / name for name in PINS}
PATHS['successor-health.py'] = HERE.parent / 'gpu-iommu-health-r1/successor-health.py'
PHASES = ('observe_target', 'post_capture_health', 'observe_fallback', 'verify_fallback_restoration', 'locate_fallback', 'verify_target_restoration')


def load(name, filename):
    import hashlib
    path = PATHS[filename]
    if (path.resolve() != path or path.stat().st_size > 65536
            or hashlib.sha256(path.read_bytes()).hexdigest() != PINS[filename]):
        raise ValueError('health callback producer changed: ' + filename)
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


R = load('health_read_common', 'receipt-common.py')
A = R.A
H = load('health_read_target', 'successor-health.py')
F = load('health_read_fallback', 'fallback-observation.py')
need = A.need


def pins():
    for name, expected in PINS.items():
        path = PATHS[name]
        need(path.resolve() == path and path.stat().st_size <= 65536
             and A.sha(path.read_bytes()) == expected, 'health producer changed')
    R.pins()
    A.pinned_sources()
    H.inputs()


DISCOVERY = r'''set -eu
export LC_ALL=C
bb() { /run/initramfs/lib/ld-musl-aarch64.so.1 /run/initramfs/bin/busybox "$@"; }
before=$(bb cat /proc/sys/kernel/random/boot_id)
release=$(bb uname -r)
bundle= count=0
set -f
for arg in $(bb cat /proc/cmdline); do
 case "$arg" in rog5.bundle=*) bundle=${arg#rog5.bundle=}; count=$((count+1));; esac
done
[ "$count" = 1 ]
ready=wait
if [ -f /run/rog5-p2-ready ] && [ ! -L /run/rog5-p2-ready ] &&
   /usr/bin/systemctl is-active --quiet rog5-persistent-ssh-identity.service; then
 if [ "$role" = fallback ] || /usr/bin/systemctl is-active --quiet rog5-wifi-healthy.service; then ready=ready; fi
fi
after=$(bb cat /proc/sys/kernel/random/boot_id)
[ "$before" = "$after" ]
printf 'boot_id=%s\nbundle=%s\nrelease=%s\nready=%s\n' "$before" "$bundle" "$release" "$ready"
'''


def discovery(role):
    need(role in ('target', 'fallback'), 'discovery role')
    return 'role=' + role + '\n' + DISCOVERY


class HealthReads:
    def __init__(self, source, authorize, transport=None):
        need(transport is None or callable(transport), 'transport callback required')
        self.transport = transport
        need(type(source) is dict and source.get('clean') is True and callable(authorize),
             'clean source and complete admission required')
        self.source = A.decode(json.dumps(source))
        self.authorize = authorize
        self.deployed = A.W.load_deployed()

    def gate(self, phase, context, results, pin):
        pins()
        need(phase in PHASES and os.geteuid() == 1000, 'health callback phase/owner')
        need(A.exact_json(context['source'], dict(A.C.SOURCE, boot_id=A.SOURCE_BOOT))
             and context['owner'] == A.OWNER, 'health source/custody binding')
        need(self.deployed.CAPTURE.ACCEPTANCE.source_identity() == self.source,
             'host source changed')
        need(self.authorize(phase, context, results, pin) is True, 'health outer admission refused')
        need(OUTPUT.resolve() == OUTPUT and OUTPUT.is_dir() and OUTPUT.stat().st_uid == 1000
             and stat.S_IMODE(OUTPUT.stat().st_mode) == 0o700, 'health output owner')
        need(A.exact_json(A.decode(R.receipt(OUTPUT / 'context.json', 4096)), context), 'health context')
        raw = R.receipt(OUTPUT / (phase + '-entered.json'), 16384)
        need(A.sha(raw) == pin, 'health intent hash')
        intent = A.decode(raw)
        need(set(intent) == {'phase', 'mutation', 'admission_sha256', 'owner', 'monotonic', 'prior'}
             and intent['phase'] == phase and intent['mutation'] is False
             and intent['owner'] == context['owner']
             and intent['admission_sha256'] == context['admission_sha256']
             and A.exact_json(intent['prior'], list(results)), 'health intent binding')
        need(type(intent['monotonic']) in (int, float) and math.isfinite(intent['monotonic'])
             and 0 < intent['monotonic'] <= time.monotonic(), 'health intent time')
        # Prior observations must be the controller's actual durable receipts.
        for name, value in results.items():
            need(name in A.C.PHASES and A.exact_json(
                A.decode(R.receipt(OUTPUT / (name + '-result.json'), 1048576)), value),
                'health prior result changed')
        return intent

    def command(self, phase, label, role, script, remote, timeout, context, results, pin):
        self.gate(phase, context, results, pin)
        mode = 'normal' if role == 'target' else 'linklocal'
        request = A.W.request(json.dumps(dict(format='rog5-one-usb-command-v1', mode=mode,
            remote=remote, script=script, source=self.source, timeout=timeout)).encode())
        prefix = OUTPUT / (phase + '-' + label)
        A.C.save(prefix.with_name(prefix.name + '-entered.json'), dict(intent_sha256=pin,
            script_sha256=A.sha(script.encode()), read_only=True, mode=mode))
        reply = (self.transport(phase, context, results, pin, request, self.deployed)
                 if self.transport is not None else A.W.perform(request, self.deployed))
        A.save_transport(prefix.with_name(prefix.name + '-transport.json'), reply)
        need(reply.get('status') == 'PASS_TRANSPORT_COMPLETED'
             and reply.get('mode') == mode and reply.get('command_invoked') is True
             and A.exact_json(reply.get('source'), self.source), 'health transport failed')
        command = reply.get('command')
        need(type(command) is dict, 'health command absent')
        out = base64.b64decode(command['stdout_base64'], validate=True)
        err = base64.b64decode(command['stderr_base64'], validate=True)
        need(A.sha(out) == command['stdout_sha256'] and A.sha(err) == command['stderr_sha256'],
             'health raw reply hashes')
        need(type(command['returncode']) is int and command['timed_out'] is False
             and command['reaped'] is True, 'health command timeout or not reaped')
        self.gate(phase, context, results, pin)
        return command['returncode'], out, err

    def previous_identity(self, phase, results):
        if phase == 'post_capture_health':
            need(results.get('close_capture', {}).get('cleanup_complete') is True
                 and results['close_capture'].get('full_lifetime') is True
                 and results['close_capture'].get('capture_status') == 'PASS', 'full capture not closed')
            return results['observe_target']['identity']
        if phase in ('verify_fallback_restoration', 'verify_target_restoration'):
            value = results['restore_target_state' if phase=='verify_target_restoration' else 'restore_fallback_state']
            need(value.get('completed') is True and value.get('after_sha256') == A.G.OLD_SHA,
                 'fallback restore incomplete')
            return value['identity']
        return None

    def observe(self, phase, context, results, pin):
        intent = self.gate(phase, context, results, pin)
        role = 'target' if phase in ('observe_target', 'post_capture_health', 'verify_target_restoration') else 'fallback'
        expected = A.C.TARGET if role == 'target' else A.C.FALLBACK
        excluded = [A.SOURCE_BOOT]
        if role == 'fallback' and 'observe_target' in results:
            excluded.append(results['observe_target']['identity']['boot_id'])
        previous = self.previous_identity(phase, results)
        if previous is not None:
            A.C.same_identity(previous, expected, excluded=excluded)
        A.C.save(OUTPUT / (phase + '-command-entered.json'), dict(intent_sha256=pin, read_only=True))
        # Bounded readiness polling only for initial observations. A later check
        # must already be ready on the previously observed boot.
        deadline = intent['monotonic'] + 300
        initial = previous is None
        found = None
        for attempt in range(150 if initial else 1):
            remaining = deadline - time.monotonic()
            need(remaining >= 1, 'initial health observation deadline expired')
            self.gate(phase, context, results, pin)
            cap = self.deployed.CAPTURE
            try:
                usb = cap.usb_mode(A.W.SERIAL)
            except cap.UsbReadDisappeared:
                usb = ('enumerating', None)
            A.C.save(OUTPUT / (phase + f'-usb-{attempt:03}.json'),
                     dict(mode=usb[0], interface=usb[1], read_only=True))
            # fastboot boot/reboot returns before USB necessarily disconnects.
            # Wait only after this run's successful one-use boot command, before
            # any authenticated target identity has been seen. A later return
            # to fastboot remains a failure, and the original deadline stands.
            boot_result = results.get('boot_once', {})
            reboot_result = results.get('reboot_installed', {})
            boot_transition = initial and found is None and (
                (phase == 'observe_target' and boot_result.get('status') == 'PASS'
                 and boot_result.get('transfer_completed') is True
                 and boot_result.get('attempt_count') == 1)
                or (phase == 'observe_fallback' and reboot_result.get('status') == 'PASS'
                    and reboot_result.get('reboot_requested') is True
                    and reboot_result.get('command_completed') is True))
            if (usb in (('absent', None), ('enumerating', None), ('recovery', None))
                    or (usb == ('fastboot', None) and boot_transition)):
                need(initial and deadline - time.monotonic() >= 2, 'health USB not ready')
                time.sleep(2)
                continue
            need(usb == ('target', cap.INTERFACE), 'health USB identity/mode mismatch')
            if phase == 'locate_fallback':
                # Probe only authenticated read-only discovery. Full health below
                # selects the actual exact kernel, never inferring it from USB.
                candidates = (role,) if found is not None else ('target', 'fallback')
                for route_role in candidates:
                    remaining = deadline-time.monotonic()
                    need(remaining>=1, 'initial health observation deadline expired')
                    code,out,err=self.command(phase,f'discovery-{attempt:03}-{route_role}',
                        route_role,discovery(route_role),A.SHELL,min(10,int(remaining)),context,results,pin)
                    if code==255 and not out:
                        continue
                    need(code==0 and not err and 0<len(out)<=1024, 'health discovery failed')
                    located=A.fields(out)
                    role=self.location_role(located,results)
                    expected=A.C.TARGET if role=='target' else A.C.FALLBACK
                    excluded=[A.SOURCE_BOOT]
                    if role=='fallback' and 'observe_target' in results:
                        excluded.append(results['observe_target']['identity']['boot_id'])
                    break
            else:
                code, out, err = self.command(phase, f'discovery-{attempt:03}', role, discovery(role),
                    A.SHELL, min(10, int(remaining)), context, results, pin)
            if code == 255 and not out and initial:
                pass  # Authenticated SSH has not completed; this is not a boot proof.
            else:
                need(code == 0 and not err and 0 < len(out) <= 1024, 'health discovery failed')
                value = A.fields(out)
                need(set(value) == {'boot_id', 'bundle', 'release', 'ready'}
                     and value['ready'] in ('ready', 'wait'), 'health discovery schema')
                identity = {k: value[k] for k in ('boot_id', 'bundle', 'release')}
                A.C.same_identity(identity, expected, excluded=excluded)
                need(previous is None or A.exact_json(identity, previous), 'health boot changed after capture/restore')
                need(found is None or A.exact_json(identity, found), 'health boot changed while waiting')
                found = identity
                if value['ready'] == 'ready':
                    break
                need(initial, 'health no longer ready')
            need(initial and deadline - time.monotonic() >= 2, 'health readiness deadline expired')
            time.sleep(2)
        else:
            raise ValueError('health readiness attempt bound')
        need(found is not None, 'health identity absent')
        boot = found['boot_id']
        purpose = 'verify-restored' if phase in ('verify_fallback_restoration','verify_target_restoration') else 'observe'
        full_target_health = role=='target' and phase!='verify_target_restoration'
        script = H.script('target', boot) if full_target_health else F.build(purpose, boot, role)
        code, out, err = self.command(phase, 'health', role, script,
            'python3 -I -B -' if full_target_health else A.SHELL, 35, context, results, pin)
        need(code == 0 and not err and 0 < len(out) <= 65536, 'health observation command failed')
        proof = H.validate(A.decode(out), 'target', boot) if full_target_health else F.parse(out, purpose, boot, role)
        need(A.exact_json(proof['identity'], found), 'health proof identity changed')
        pins()
        return dict(proof, authenticated=True, read_only=True, **({'mode':role} if phase=='locate_fallback' else {}))

    def location_role(self, value, results):
        need(set(value)=={'boot_id','bundle','release','ready'} and value['ready'] in ('ready','wait'),
             'health discovery schema')
        identity={k:value[k] for k in ('boot_id','bundle','release')}
        role = 'target' if (value['bundle'],value['release']) == (A.C.TARGET['bundle'],A.C.TARGET['release']) else 'fallback'
        expected=A.C.TARGET if role=='target' else A.C.FALLBACK
        excluded=[A.SOURCE_BOOT]
        if role=='fallback' and 'observe_target' in results:
            excluded.append(results['observe_target']['identity']['boot_id'])
        A.C.same_identity(identity,expected,excluded=excluded)
        if role=='target' and 'observe_target' in results:
            need(A.exact_json(identity,results['observe_target']['identity']), 'target recovery boot changed')
        return role

    def verify_target_restoration(self, context, results, pin):
        return self.observe('verify_target_restoration',context,results,pin)

    def observe_target(self, context, results, pin):
        return self.observe('observe_target', context, results, pin)

    def post_capture_health(self, context, results, pin):
        return self.observe('post_capture_health', context, results, pin)

    def observe_fallback(self, context, results, pin):
        return self.observe('observe_fallback', context, results, pin)

    def verify_fallback_restoration(self, context, results, pin):
        return self.observe('verify_fallback_restoration', context, results, pin)

    def locate_fallback(self, context, results, pin):
        return self.observe('locate_fallback', context, results, pin)
