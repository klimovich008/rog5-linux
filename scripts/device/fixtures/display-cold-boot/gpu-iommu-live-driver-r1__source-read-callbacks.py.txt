HERE=Path(__file__).resolve().parent
PINS={'action-callbacks.py':'f655b7a39ff599d593393cff41fbad33198486d550bf000f82dd15577cde314e',
      'generate-source-observation.py':'02d7976fdf93d28e396c9680ebe36bd19d1fcbb9d42dd70694847a13aaf7430a'}
PHASES={'preflight':'preflight','inspect_source_abort':'inspect-source-abort','verify_source_abort':'verify-source-abort'}
OUTPUT=HERE.parent/'gpu-iommu-controller-r1/execution'


def pins():
    for name,digest in PINS.items():
        p=HERE/name
        if p.resolve()!=p or p.stat().st_size>65536 or hashlib.sha256(p.read_bytes()).hexdigest()!=digest:
            raise ValueError('read callback producer changed: '+name)


def load(name,path):
    s=importlib.util.spec_from_file_location(name,path);m=importlib.util.module_from_spec(s);s.loader.exec_module(m);return m

def validate(value,purpose,context):
    need(purpose in PHASES.values(),'source observation purpose')
    need(type(value) is dict,'source observation object')
    need(context['owner']==A.OWNER and context['source']==dict(A.C.SOURCE,boot_id=A.SOURCE_BOOT),
         'source observation context')
    state=value.get('state_sha256');shutdown=value.get('shutdown');kind=value.get('selection_kind')
    need(state in (A.G.OLD_SHA,A.G.PENDING_SHA) and shutdown in ('original','patched'),'source state/shutdown')
    need(kind in (('absent','restored') if state==A.G.OLD_SHA else ('staged',)),'source transaction kind')
    if shutdown=='patched':need(kind=='staged','patched source selection')
    original='fc1ce027c20679a1d18200858e216c106c0fa690224de26e3705c105fda67860'
    patched='0ee47fb596eb4ba7d6df63b6632a04d81524e0d4c09cdbb859dba698cd9684ca'
    expected=dict(status='PASS_SOURCE_OBSERVATION',
        purpose='verify-source-abort' if purpose=='preflight' else purpose,
        boot_id=A.SOURCE_BOOT,owner=A.OWNER,selection_kind=kind,state_sha256=state,shutdown=shutdown,
        shutdown_sha256=original if shutdown=='original' else patched,reboot_intent_absent='true',
        physical_guards_passed='true',read_only='true')
    need(A.exact_json(value,expected),'source observation proof mismatch')
    if purpose=='preflight':need(kind=='absent' and shutdown=='original','source not pristine')
    if purpose=='verify-source-abort':need(state==A.G.OLD_SHA and shutdown=='original','source abort incomplete')
    return dict(format='rog5-source-transition-observation-v2',status='PASS',purpose=purpose,
        identity=context['source'],owner=context['owner'],state_sha256=state,shutdown=shutdown,
        shutdown_sha256=expected['shutdown_sha256'],selection_kind=kind,raw_observation=value,
        reboot_intent_absent=True,shutdown_restored=shutdown=='original',ram_staging_verified=True,
        physical_guards_passed=True,read_only=True,release_qualified=False)


def receipt(path,limit):
    need(path.resolve()==path,'canonical source receipt')
    fd=os.open(path,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK|os.O_CLOEXEC)
    def metadata(st):return (st.st_dev,st.st_ino,st.st_mode,st.st_uid,st.st_gid,st.st_nlink,st.st_size,st.st_mtime_ns,st.st_ctime_ns)
    try:
        st=os.fstat(fd)
        need(stat.S_ISREG(st.st_mode) and st.st_uid==1000 and st.st_nlink==1
             and stat.S_IMODE(st.st_mode)==0o600 and 0<st.st_size<=limit,'source receipt metadata')
        raw=os.read(fd,limit+1)
        need(len(raw)==st.st_size and not os.read(fd,1) and metadata(st)==metadata(os.fstat(fd))==metadata(path.lstat()),'source receipt changed')
        return raw
    finally:os.close(fd)


class SourceReads:
    def __init__(self,source,authorize):
        need(callable(authorize) and type(source) is dict and source.get('clean') is True,'outer source/admission required')
        self.source=A.decode(json.dumps(source));self.authorize=authorize;self.deployed=A.W.load_deployed()

    def observe(self,phase,context,results,pin):
        need(phase in PHASES and os.geteuid()==1000,'source read phase/owner')
        pins();A.pinned_sources()
        need(context['owner']==A.OWNER and context['source']==dict(A.C.SOURCE,boot_id=A.SOURCE_BOOT),'fixed staged source')
        need(self.deployed.CAPTURE.ACCEPTANCE.source_identity()==self.source,'host source changed')
        need(self.authorize(phase,context,results,pin) is True,'outer source read admission refused')
        need(OUTPUT.resolve()==OUTPUT and OUTPUT.stat().st_uid==1000
             and stat.S_IMODE(OUTPUT.stat().st_mode)==0o700,'source read output')
        need(A.exact_json(A.decode(receipt(OUTPUT/'context.json',4096)),context),'source read context')
        raw=receipt(OUTPUT/(phase+'-entered.json'),16384);need(A.sha(raw)==pin,'source read intent hash')
        intent=A.decode(raw)
        need(set(intent)=={'phase','mutation','admission_sha256','owner','monotonic','prior'}
             and intent['phase']==phase and intent['mutation'] is False and intent['owner']==A.OWNER
             and intent['admission_sha256']==context['admission_sha256'] and intent['prior']==list(results),'source read intent binding')
        script=G.build(PHASES[phase],A.SOURCE_BOOT,A.OWNER,A.RECEIPT_SHA)
        request=A.W.request(json.dumps(dict(format='rog5-one-usb-command-v1',mode='normal',remote=A.SHELL,
                         script=script,source=self.source,timeout=35)).encode())
        A.C.save(OUTPUT/(phase+'-command-entered.json'),dict(intent_sha256=pin,script_sha256=A.sha(script.encode()),read_only=True))
        reply=A.W.perform(request,self.deployed)
        A.save_transport(OUTPUT/(phase+'-transport.json'),reply)
        need(reply.get('status')=='PASS_TRANSPORT_COMPLETED' and reply.get('mode')=='normal'
             and reply.get('command_invoked') is True and A.exact_json(reply.get('source'),self.source),
             'source read transport provenance')
        c=reply.get('command');need(type(c) is dict,'source read command missing')
        out=base64.b64decode(c['stdout_base64'],validate=True);err=base64.b64decode(c['stderr_base64'],validate=True)
        need(A.sha(out)==c['stdout_sha256'] and A.sha(err)==c['stderr_sha256'],'source read raw hashes')
        need(type(c['returncode']) is int and c['returncode']==0 and c['timed_out'] is False and c['reaped'] is True
             and not err and 0<len(out)<=16384,'source read command failed or unbounded')
        pins();A.pinned_sources()
        return validate(A.fields(out),PHASES[phase],context)

    def preflight(self,context,results,pin):return self.observe('preflight',context,results,pin)
    def inspect_source_abort(self,context,results,pin):return self.observe('inspect_source_abort',context,results,pin)
    def verify_source_abort(self,context,results,pin):return self.observe('verify_source_abort',context,results,pin)
