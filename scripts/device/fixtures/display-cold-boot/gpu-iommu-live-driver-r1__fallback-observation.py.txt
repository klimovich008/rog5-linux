No helper invocation, state publication, reboot, Python or target mutation.
The host must bind a fresh SSH peer/boot and the complete controller admission.
"""

def need(ok,why):
    if not ok:raise ValueError(why)


def sha(raw):return hashlib.sha256(raw).hexdigest()


def load(name,path):
    s=importlib.util.spec_from_file_location(name,path);m=importlib.util.module_from_spec(s);s.loader.exec_module(m);return m


need(sha((GUARDS/'guarded-state.py').read_bytes())=='c4831060d80f379878cb241e384d55948f0b46e66e021a639338f72dffa901ad','physical guard changed')
G=load('fallback_observation_guard',GUARDS/'guarded-state.py')


def inventory():
    # Installed V9/V11 files are unchanged. Reuse their pinned inventory only;
    # the new route query must verify these bytes on the current source.
    receipt=(STATE/'successor-state-guards-r1/custody-<OWNER_2>/receipt.json').read_bytes()
    need(sha(receipt)=='1dd818c93235a8cfd1017f73339f591bffcb843c95ff1a1f56ac43014104468c','host custody changed')
    bound=json.loads(receipt)
    path=Path(bound['health_directory'])/'request.json'
    need(path.resolve()==path and path.stat().st_size<=16384,'installed request bound')
    raw=path.read_bytes();need(sha(raw)==bound['health_sha256']['request.json'],'installed request changed')
    value=json.loads(raw);need(value['boot_id']=='<BOOT_UUID_2>' and sha(value['healthy_trial'].encode())=='ed3a62d198a2e33d26093e079534315ae86ab28c91f48805463fd025c5b88b75','installed source identity')
    files=value['files']
    need(len(files)==11,'installed inventory size')
    prefix='/.rog5/root-ro/boot/rog5-linux/'
    names={prefix+'selector'}
    for bundle in ('headless-server-selector-v9','persistent-native-root-v11'):
        names.update(prefix+'bundles/'+bundle+'/'+name for name in ('Image','board.dtb','initramfs.cpio.gz','manifest','manifest.sig'))
    need(set(files)==names,'installed inventory members')
    return files


BODY=r'''
service_active() { /usr/bin/systemctl is-active rog5-persistent-ssh-identity.service; }
ready_snapshot() {
 regular /run/rog5-p2-ready 444 16384
 [ -d /run ] && [ ! -L /run ] && [ "$(bb stat -f -c %T /run/rog5-p2-ready)" = tmpfs ] || fail 'readiness not in RAM'
 [ "$(service_active)" = active ] || fail 'SSH identity service inactive'
 ready_stamp=$(bb stat -c '%d:%i:%s:%y:%z:%u:%g:%a:%h' /run/rog5-p2-ready)
 ready_hash=$(bb sha256sum /run/rog5-p2-ready | bb cut -d ' ' -f 1)
 ready_text=$(bb base64 /run/rog5-p2-ready | bb tr -d '\n')
 [ "$ready_stamp" = "$(bb stat -c '%d:%i:%s:%y:%z:%u:%g:%a:%h' /run/rog5-p2-ready)" ] || fail 'readiness changed'
}
restoration_snapshot() {
 tx=$state/rog5/boot/.gpu-iommu-136f7-transition
 [ -d "$tx" ] && [ ! -L "$tx" ] && [ "$(bb stat -c '%u:%g:%a' "$tx")" = 0:0:700 ] || fail 'restoration transaction directory'
 count=0
 for path in "$tx"/* "$tx"/.[!.]* "$tx"/..?*; do
  [ -e "$path" ] || [ -L "$path" ] || continue
  case ${path##*/} in stage-intent|stage-completed|old-backup|exchange|restore-intent|restore-completed) ;; *) fail 'unknown restoration member';; esac
  regular "$path" 600 512;count=$((count+1))
 done
 [ "$count" = 5 ] || [ "$count" = 6 ] || fail 'partial restoration inventory'
 hash_is "$tx/stage-intent" "$stage_token_hash"
 if [ -e "$tx/stage-completed" ]; then hash_is "$tx/stage-completed" "$stage_token_hash"; fi
 hash_is "$tx/old-backup" "$old_hash"
 hash_is "$tx/restore-intent" "$restore_token_hash"
 hash_is "$tx/restore-completed" "$restore_token_hash"
 exchanged=$(bb sha256sum "$tx/exchange" | bb cut -d ' ' -f 1)
 case $exchanged in "$pending_hash"|"$healthy_hash") ;; *) fail 'restoration exchange record';; esac
 parent=${helper%/*}
 [ -d "$parent" ] && [ ! -L "$parent" ] && [ "$(bb stat -c '%u:%g:%a' "$parent")" = 0:0:700 ] || fail 'restoration RAM directory'
 regular "$helper" 755 2097152;hash_is "$helper" "$helper_hash"
 regular "$custody_file" 400 1024;hash_is "$custody_file" "$custody_hash"
 regular "$parent/staging-completed" 400 1024;hash_is "$parent/staging-completed" "$custody_hash"
 [ -d "$parent/.$restore_operation-entered" ] && [ ! -L "$parent/.$restore_operation-entered" ] &&
 [ "$(bb stat -c '%u:%g:%a' "$parent/.$restore_operation-entered")" = 0:0:700 ] || fail 'restoration operation not entered'
 regular "$parent/$restore_operation.stdout" 600 4096;hash_is "$parent/$restore_operation.stdout" "$restore_reply_hash"
 regular "$parent/$restore_operation.stderr" 600 4096
 [ "$(bb stat -c %s "$parent/$restore_operation.stderr")" = 0 ] || fail 'restoration helper stderr'
}

guard
regular "$record" 600 512
state_stamp=$(bb stat -c '%d:%i:%s:%y:%z:%u:%g:%a:%h' "$record")
state_hash=$(bb sha256sum "$record" | bb cut -d ' ' -f 1)
case $purpose:$state_hash in
 observe:$pending_hash|observe:$healthy_hash) ;;
 verify-restored:$old_hash) restoration_snapshot;;
 *) fail 'wrong fallback observation state';;
esac
ready_snapshot
first_ready_hash=$ready_hash
first_ready_stamp=$ready_stamp
'''


def operation(role,purpose):
    need(role in ('fallback','target'), 'restoration observation role')
    need(role != 'target' or purpose == 'verify-restored', 'target observation requires full health reader')
    return 'restore-target' if role == 'target' else 'restore-v11'


def build(purpose,boot,role='fallback'):
    need(purpose in ('observe','verify-restored'),'fallback observation purpose')
    G.boot_id(boot);need(boot!=SOURCE_BOOT,'source boot is not fallback')
    op=operation(role,purpose)
    full=G.build(op,boot,SOURCE_BOOT,OWNER,RECEIPT_SHA)
    need(full.count('run_helper() {')==1,'physical guard boundary')
    script=full.split('run_helper() {')[0]
    fields=dict(purpose=purpose,restore_operation=op,stage_token_hash=sha(b'rog5-gpuiommu136-stage-v1\n'),restore_token_hash=sha(b'rog5-gpuiommu136-restore-v1\n'),
                restore_reply_hash=sha(b'completed=restore\n'))
    script+=''.join(k+'='+shlex.quote(v)+'\n' for k,v in fields.items())+BODY
    script+='''
[ -b /dev/sde35 ] && [ ! -L /dev/sde35 ] || fail 'installed boot device type'
[ "$(bb readlink -f /dev/disk/by-partlabel/boot_b)" = /dev/sde35 ] || fail 'installed boot label'
[ "$(bb cat /sys/class/block/sde35/ro)" = 1 ] && [ "$(bb cat /sys/class/block/sde35/size)" = 196608 ] || fail 'installed boot geometry/protection'
hash_is /dev/sde35 '''+shlex.quote(BOOT_SHA)+'\n'
    files=inventory()
    for path,row in sorted(files.items()):
        # Check each parent without following symlinks, including the read-only
        # boot tree. Actual block contents are hashed by BusyBox in a stream.
        for parent in reversed(Path(path).parents):
            if str(parent)=='/':continue
            q=shlex.quote(str(parent));script+='[ -d '+q+' ] && [ ! -L '+q+" ] || fail 'installed parent path'\n"
        q=shlex.quote(path);script+=f'regular {q} {row["mode"]:o} {row["limit"]};hash_is {q} {row["sha256"]}\n'
    script+='''
guard
regular "$record" 600 512;hash_is "$record" "$state_hash"
[ "$state_stamp" = "$(bb stat -c '%d:%i:%s:%y:%z:%u:%g:%a:%h' "$record")" ] || fail 'selection changed during observation'
[ "$purpose" != verify-restored ] || restoration_snapshot
ready_snapshot
[ "$first_ready_hash" = "$ready_hash" ] && [ "$first_ready_stamp" = "$ready_stamp" ] || fail 'readiness changed during observation'
printf 'format=rog5-v11-selection-observation-v1\\npurpose=%s\\nboot_id=%s\\nstate_sha256=%s\\nreadiness_sha256=%s\\nreadiness_base64=%s\\ninstalled_inventory_sha256=%s\\nphysical_guards_passed=true\\nrestoration_verified=%s\\nread_only=true\\n' "$purpose" "$expected_boot" "$state_hash" "$ready_hash" "$ready_text" '''
    script+=shlex.quote(sha(json.dumps(dict(files,**{'/dev/sde35':dict(sha256=BOOT_SHA)}),sort_keys=True,separators=(',',':')).encode()))
    script+=' "$(if [ "$purpose" = verify-restored ]; then printf true; else printf false; fi)"\n'
    if role=='target':
        need(script.count('rog5-v11-selection-observation-v1')==1,'restoration format boundary')
        script=script.replace('rog5-v11-selection-observation-v1','rog5-target-selection-restoration-v1')
    return script


def parse(raw,purpose,boot,role='fallback'):
    op=operation(role,purpose)
    bundle,release,_=G.OPERATIONS[op]
    expected_format='rog5-target-selection-restoration-v1' if role=='target' else 'rog5-v11-selection-observation-v1'
    need(type(raw) is bytes and 0<len(raw)<=32768 and raw.endswith(b'\n'),'fallback reply bound/framing')
    rows=[line.split('=',1) for line in raw.decode('ascii').splitlines()]
    need(all(len(row)==2 for row in rows) and len({row[0] for row in rows})==len(rows),'fallback reply fields')
    fields=dict(rows)
    need(set(fields)=={'format','purpose','boot_id','state_sha256','readiness_sha256','readiness_base64',
         'installed_inventory_sha256','physical_guards_passed','restoration_verified','read_only'},'fallback reply schema')
    need(purpose in ('observe','verify-restored') and boot!=SOURCE_BOOT,'fallback purpose/boot');G.boot_id(boot)
    need(fields['format']==expected_format and fields['purpose']==purpose and fields['boot_id']==boot
         and fields['physical_guards_passed']==fields['read_only']=='true'
         and fields['restoration_verified']==('true' if purpose=='verify-restored' else 'false'),'fallback proof identity')
    need(fields['state_sha256'] in ((G.OLD_SHA,) if purpose=='verify-restored' else (G.PENDING_SHA,G.HEALTHY_SHA)),'fallback selection proof')
    expected=sha(json.dumps(dict(inventory(),**{'/dev/sde35':dict(sha256=BOOT_SHA)}),sort_keys=True,separators=(',',':')).encode())
    need(fields['installed_inventory_sha256']==expected,'fallback installed inventory proof')
    marker=base64.b64decode(fields['readiness_base64'],validate=True)
    need(0<len(marker)<=16384 and sha(marker)==fields['readiness_sha256'],'fallback readiness digest')
    # The exact signed V11 producer has a legacy marker. A supplied boot ID
    # must match; its absence is explicitly reported, never fabricated.
    need(marker.endswith(b'\n') and b'\r' not in marker and b'\0' not in marker,'fallback marker framing')
    rows=[line.split('=',1) for line in marker.decode('ascii').splitlines()]
    need(all(len(row)==2 for row in rows) and len({row[0] for row in rows})==len(rows),'fallback marker fields')
    ready=dict(rows)
    need(ready.get('status')=='PASS' and ready.get('kernel')==release
         and ready.get('ssh')=='strict-key-only','fallback readiness failed')
    need('attested_boot_id' not in ready or ready['attested_boot_id']==boot,'stale fallback readiness')
    need(role!='target' or ready.get('attested_boot_id')==boot,'target readiness not boot-bound')
    return dict(status='PASS',identity=dict(boot_id=boot,bundle=bundle,release=release),
                state_sha256=fields['state_sha256'],physical_guards_passed=True,read_only=True,
                readiness_boot_bound='attested_boot_id' in ready,installed_inventory_verified=True,
                selection_eligibility_restored=purpose=='verify-restored',release_qualified=False)
