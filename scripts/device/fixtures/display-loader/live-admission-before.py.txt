def need(value, message):
    if not value:
        raise ValueError(message)


def stamp(st):
    return (st.st_dev, st.st_ino, st.st_uid, st.st_gid, st.st_mode,
            st.st_nlink, st.st_size, st.st_mtime_ns, st.st_ctime_ns)


def digest_file(path, size, uid, mode, expected):
    """Stream once per unchanged inode; metadata changes invalidate the cache."""
    need(path.resolve() == path, 'admission input is not canonical')
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK | os.O_CLOEXEC)
    try:
        st = os.fstat(fd)
        need(stat.S_ISREG(st.st_mode) and st.st_uid == uid and st.st_nlink == 1
             and stat.S_IMODE(st.st_mode) == mode and st.st_size == size,
             'admission input metadata: ' + path.name)
        key = (str(path), stamp(st), expected)
        if key not in _CACHE:
            h = hashlib.sha256()
            with os.fdopen(os.dup(fd), 'rb') as stream:
                for block in iter(lambda: stream.read(1048576), b''):
                    h.update(block)
            need(h.hexdigest() == expected, 'admission input digest: ' + path.name)
        need(stamp(st) == stamp(os.fstat(fd)) == stamp(path.lstat()), 'admission input changed while reading')
        _CACHE[key] = True
    finally:
        os.close(fd)


def load_driver():
    p = HERE / 'live-driver.py'
    need(p.resolve() == p and p.stat().st_size < 262144
         and hashlib.sha256(p.read_bytes()).hexdigest() == DRIVER_SHA, 'admission driver changed')
    s = importlib.util.spec_from_file_location('admitted_driver', p)
    m = importlib.util.module_from_spec(s); s.loader.exec_module(m)
    return m

def inputs():
    need(type(INPUTS_SHA) is str and re.fullmatch('[0-9a-f]{64}',INPUTS_SHA),'live input closure not prepared')
    D.BOOT.expected_kernel_relay()
    raw = R.receipt(INPUTS, 131072)
    need(A.sha(raw) == INPUTS_SHA, 'admission input lock changed')
    value = A.decode(raw)
    need(set(value) == {'format', 'files'} and value['format'] == 'rog5-live-admission-inputs-v1', 'input lock schema')
    for name, row in value['files'].items():
        need(not Path(name).is_absolute() and '..' not in Path(name).parts
             and set(row) == {'sha256', 'size', 'uid', 'mode'}, 'input lock row')
        digest_file(STATE / name, row['size'], row['uid'], row['mode'], row['sha256'])
    D.pins(); D.BOOT.pins(); W.pins(); A.pinned_sources(); H.pins()
    # Existing consumed RAM-stage receipts remain 0600 and are never restaged.
    for name, expected in [('preparation.json', A.RECEIPT_SHA),
                           ('stage-receipt.json', 'f59cb4e2819f17c1430724bd0f83d913f9c763897091d7147f8ea269101c2681')]:
        p = A.STAGING / name
        need(p.stat().st_size <= 16384, 'custody size')
        digest_file(p, p.stat().st_size, 1000, 0o600, expected)
    need(A.sha(A.G.custody(A.OWNER, A.SOURCE_BOOT, A.RECEIPT_SHA)) == A.CUSTODY_SHA,
         'GPU custody identity changed')
    return value


def claim():
    """The lifecycle implementation is account-relative: always verify as deck."""
    if os.geteuid() == 1000:
        return B.canonical(D.BOOT.IMAGE_SHA, B.SERIAL)
    need(os.geteuid() == 0, 'admission claim owner')
    code = ('import importlib.util,json,sys\n'
            'sys.path.insert(0,' + repr(str(D.BOOT.REPO / 'scripts/host')) + ')\n'
            's=importlib.util.spec_from_file_location("claim_check",' +
            repr(str(D.BOOT.REPO / 'scripts/host/verified-gpu-iommu-hardware-boot.py')) + ')\n'
            'm=importlib.util.module_from_spec(s);s.loader.exec_module(m)\n'
            'print(json.dumps(m.canonical(' + repr(D.BOOT.IMAGE_SHA) + ',' + repr(B.SERIAL) + ')))\n')
    result = A.W.execute(['/usr/bin/runuser', '-u', 'deck', '--', '/usr/bin/python3', '-I', '-B', '-c', code], '', 10)
    out, err = D.BOOT.raw_result(result)
    need(not err, 'claim verifier stderr')
    return A.decode(out)


def qualification(source):
    value = receipt(QUALIFICATION)
    expected = dict(format='rog5-kernel-live-qualification-v1', status='PASS',
        inputs_sha256=INPUTS_SHA, verifier_sha256=A.sha(Path(__file__).read_bytes()), host_source=source,
        launcher_sha256=A.sha(LAUNCHER.read_bytes()),
        target_recovery_verified=True, adaptive_recovery_discovery_verified=True,
        gpu_session_qualified=True, gpu_failure_cleanup_verified=True, gpu_fallback_handoff=True,
        target_success=True, source_abort_restored=True, early_fallback_restored=True,
        late_fallback_restored=True, privileged_capture_handoff=True,
        privileged_transport_handoff=True, cleanup_verified=True, no_phone_mutations=True)
    need(A.exact_json(value.get('checks'), expected), 'full live-flow/handoff qualification missing')
    evidence = value.get('evidence')
    need(set(value) == {'checks', 'evidence'} and type(evidence) is dict and bool(evidence), 'qualification evidence missing')
    for name, digest in evidence.items():
        p = HERE / name
        need(not Path(name).is_absolute() and '..' not in Path(name).parts,
             'qualification evidence path')
        need(A.sha(R.receipt(p, 1048576)) == digest, 'qualification evidence changed')
    return A.sha(R.receipt(QUALIFICATION, 131072))


def common(context, admitted=None, root_envelope=None):
    need(os.geteuid() in (0, 1000), 'admission execution owner')
    need(OUTPUT.resolve() == OUTPUT and OUTPUT.is_dir() and OUTPUT.stat().st_uid == 1000
         and stat.S_IMODE(OUTPUT.stat().st_mode) == 0o700, 'admission execution directory')
    need(A.exact_json(receipt(OUTPUT / 'context.json', 4096), context)
         and set(context) == {'source', 'owner', 'admission_sha256'}
         and A.exact_json(context['source'], dict(C.SOURCE, boot_id=A.SOURCE_BOOT))
         and context['owner'] == A.OWNER, 'admission context')
    raw = R.receipt(OUTPUT / 'admission.json', 131072)
    need(A.sha(raw) == context['admission_sha256'], 'admission receipt digest')
    value = A.decode(raw)
    if admitted is not None:
        need(A.exact_json(value, admitted), 'root admission changed')
    keys = {'format', 'status', 'host_source', 'source_phone', 'owner', 'host_boot_id', 'controller',
            'inputs_sha256', 'root_capture_verifier_sha256', 'root_transport_verifier_sha256',
            'qualification_sha256', 'claim_fields', 'created_monotonic', 'expires_monotonic', 'launcher_sha256','source_preparation_sha256'}
    need(type(value) is dict and set(value) == keys and value['format'] == 'rog5-kernel-driver-admission-v1'
         and value['status'] == 'ADMITTED' and value['owner'] == A.OWNER
         and A.exact_json(value['source_phone'], context['source']), 'admission schema/source')
    now = time.monotonic(); start = value['created_monotonic']; end = value['expires_monotonic']
    need(all(type(t) in (int, float) and math.isfinite(t) for t in (start, end))
         and 0 < start <= now < end <= start + MAX_LIFETIME, 'admission lifetime')
    need(value['host_boot_id'] == Path('/proc/sys/kernel/random/boot_id').read_text().strip(), 'admission host boot')
    owner = value['controller']
    need(type(owner) is dict and set(owner) == {'pid', 'start', 'uid'} and owner['uid'] == 1000
         and type(owner['pid']) is int and owner['pid'] > 1 and type(owner['start']) is str,
         'admission controller schema')
    actual = S.process_identity(owner['pid'])
    need(actual['start'] == owner['start'] and actual['uids'] == [1000]*4, 'admission controller changed')
    if root_envelope is None:
        need(os.geteuid() == 1000 and os.getpid() == owner['pid'], 'phase caller is not original controller')
    else:
        need(os.geteuid() == 0 and A.exact_json(root_envelope['controller'], owner)
             and root_envelope['host_boot_id'] == value['host_boot_id'], 'root caller/controller binding')
    need(value['inputs_sha256'] == INPUTS_SHA, 'admission input identity')
    own_sha = A.sha(Path(__file__).read_bytes())
    need(value['root_capture_verifier_sha256'] == own_sha == value['root_transport_verifier_sha256'], 'admission verifier identity')
    inputs()
    digest_file(LAUNCHER, LAUNCHER.stat().st_size, 1000, 0o644, value['launcher_sha256'])
    source = W.source_identity()
    need(source.get('clean') is True and A.exact_json(source, value['host_source']), 'admission host source changed')
    PREP.verify_ready(value['source_preparation_sha256'],source,value['created_monotonic'])
    need(qualification(source) == value['qualification_sha256'], 'admission qualification digest')
    need(A.exact_json(value['claim_fields'], B.expected_fields())
         and A.exact_json(claim(), value['claim_fields']), 'exact one-use claim missing')
    return value


def history(context, results, phase, pin=None):
    need(phase in C.PHASES and phase not in ('close_capture', 'close_recovery_capture'), 'admission phase')
    for name, value in results.items():
        need(name in C.PHASES and value.get('status') == 'PASS'
             and A.exact_json(receipt(OUTPUT / (name + '-result.json'), 1048576), value), 'admission prior result')
    entered = set()
    for name in C.PHASES:
        path = OUTPUT / (name + '-entered.json')
        if not os.path.lexists(path):
            continue
        raw = R.receipt(path, 16384); value = A.decode(raw)
        need(set(value) == {'phase', 'mutation', 'admission_sha256', 'owner', 'monotonic', 'prior'}
             and value['phase'] == name and value['mutation'] is (name in C.MUTATIONS)
             and value['owner'] == context['owner'] and value['admission_sha256'] == context['admission_sha256']
             and type(value['monotonic']) in (int, float) and math.isfinite(value['monotonic'])
             and 0 < value['monotonic'] <= time.monotonic(), 'admission historical intent')
        need(type(value['prior']) is list and len(value['prior']) == len(set(value['prior']))
             and all(p in results for p in value['prior']), 'admission historical prior')
        entered.add(name)
        if name == phase:
            need(pin is not None and A.sha(raw) == pin and A.exact_json(value['prior'], list(results)), 'admission current intent')
    need((phase in entered) is (pin is not None) and phase not in results, 'phase reuse or missing intent')
    need(all(name in entered for name in results), 'result without entered intent')
    policy(phase, results, entered)
    return entered


def policy(phase, results, entered):
    """Additional ordering gate; physical evidence remains checked by callbacks."""
    required = {
        'preflight': (), 'arm_state': ('preflight',), 'install_exitrd': ('arm_state',),
        'request_source_reboot': ('install_exitrd',), 'await_fastboot': ('request_source_reboot',),
        'capture_start': ('await_fastboot',), 'boot_once': ('await_fastboot', 'capture_start'),
        'observe_target': ('boot_once',), 'post_capture_health': ('observe_target', 'close_capture'),
        'inspect_source_abort': (), 'restore_source_exitrd': ('inspect_source_abort',),
        'restore_source_state': ('inspect_source_abort',), 'verify_source_abort': ('inspect_source_abort',),
        'locate_fallback': (), 'recovery_capture_start': ('locate_fallback',),
        'reboot_installed': ('locate_fallback', 'recovery_capture_start'),
        'observe_fallback': ('reboot_installed',), 'stage_fallback': ('locate_fallback',),
        'restore_fallback_state': ('stage_fallback',), 'verify_fallback_restoration': ('restore_fallback_state',),
        'stage_target_recovery': ('locate_fallback',), 'restore_target_state': ('stage_target_recovery',),
        'verify_target_restoration': ('restore_target_state',),
    }
    need(phase in required and all(p in results for p in required[phase]), 'phase prerequisites missing')
    if phase == 'preflight':
        need(not results and not entered - {'preflight'}, 'preflight is not first')
    if phase in ('inspect_source_abort', 'restore_source_exitrd', 'restore_source_state', 'verify_source_abort'):
        need('request_source_reboot' not in entered, 'source reboot may have been requested')
    if phase == 'restore_source_state':
        probe = results['inspect_source_abort']
        need(probe.get('state_sha256') == C.PENDING
             and (probe.get('shutdown') == 'original' or results.get('restore_source_exitrd', {}).get('shutdown_restored') is True),
             'source state restoration before exitrd restoration')
    if phase == 'restore_source_exitrd':
        need(results['inspect_source_abort'].get('shutdown') == 'patched', 'source exitrd restore unnecessary')
    fallback = phase in ('locate_fallback', 'recovery_capture_start', 'reboot_installed', 'observe_fallback',
                         'stage_fallback', 'restore_fallback_state', 'verify_fallback_restoration',
                         'stage_target_recovery','restore_target_state','verify_target_restoration')
    if fallback:
        need('request_source_reboot' in entered, 'fallback before source reboot intent')
        if 'capture_start' in entered:
            closed = results.get('close_capture', {})
            need(closed.get('cleanup_complete') is True
                 and (closed.get('capture_started') is False and closed.get('no_owned_process') is True
                      or closed.get('capture_started') is True and closed.get('full_lifetime') is True),
                 'fallback before target capture closure')
    if phase in ('recovery_capture_start', 'reboot_installed'):
        need(results['locate_fallback'].get('mode') == 'fastboot'
             and results['locate_fallback'].get('installed_fallback_route_verified') is True, 'fallback reboot route')
    if phase in ('stage_target_recovery','restore_target_state','verify_target_restoration'):
        proof=results['locate_fallback']
        C.same_identity(proof.get('identity'),C.TARGET,excluded=(A.SOURCE_BOOT,))
        need(proof.get('mode')=='target' and proof.get('state_sha256')==C.HEALTHY
             and all(proof.get(k) is True for k in ('authenticated','physical_guards_passed',
                 'current_boot_healthy','readiness_boot_bound')),'target recovery without full health proof')
        if 'observe_target' in results:
            need(A.exact_json(proof['identity'],results['observe_target']['identity']),'target recovery boot changed')
        need('recovery_capture_start' not in entered,'target recovery after fallback capture')
        if phase=='restore_target_state':
            staged=results['stage_target_recovery']
            need(staged.get('ram_files_verified') is True and A.exact_json(staged.get('identity'),proof['identity']),
                 'target recovery helper not staged')
        if phase=='verify_target_restoration':
            restored=results['restore_target_state']
            need(restored.get('completed') is True and restored.get('before_sha256')==C.HEALTHY
                 and restored.get('after_sha256')==C.OLD and A.exact_json(restored.get('identity'),proof['identity']),
                 'target restoration incomplete')
    if phase in ('stage_fallback', 'restore_fallback_state'):
        proof = results.get('observe_fallback', results.get('locate_fallback', {}))
        C.same_identity(proof.get('identity'), C.FALLBACK, excluded=(A.SOURCE_BOOT,))
        need(proof.get('authenticated') is True and proof.get('physical_guards_passed') is True
             and proof.get('state_sha256') in (C.PENDING, C.HEALTHY), 'fallback mutation without physical proof')
        if 'recovery_capture_start' in entered:
            closed = results.get('close_recovery_capture', {})
            need(closed.get('full_lifetime') is True and closed.get('cleanup_complete') is True
                 and closed.get('capture_status') == 'PASS', 'fallback mutation before full capture closure')


def phase(phase_name, context, results, pin=None, admitted=None, envelope=None):
    value = common(context, admitted, envelope)
    history(context, results, phase_name, pin)
    return value

def authorize(name, context, results, pin):

def discovered_boot(name, role, results):
    if name == 'post_capture_health':
        return results['observe_target']['identity']['boot_id']
    if name in ('verify_fallback_restoration','verify_target_restoration'):
        return results['restore_target_state' if name=='verify_target_restoration' else 'restore_fallback_state']['identity']['boot_id']
    def order(path):
        match=re.fullmatch(re.escape(name)+r'-discovery-([0-9]{3})(?:-(target|fallback))?-transport.json',path.name)
        need(match is not None,'health discovery filename')
        need((match[2] is not None)==(name=='locate_fallback'),'health discovery route label')
        return int(match[1]),1 if match[2]=='fallback' else 0
    candidates=sorted(OUTPUT.glob(name+'-discovery-*-transport.json'),key=order)
    need(0<len(candidates)<=(300 if name=='locate_fallback' else 150),'health discovery evidence absent')
    route_role=(re.search(r'-(target|fallback)-transport.json$',candidates[-1].name)[1]
                if name=='locate_fallback' else role)
    raw = receipt(candidates[-1], 4 * 1024**2)
    need(raw.get('status') == 'PASS_TRANSPORT_COMPLETED' and raw.get('command_invoked') is True
         and raw.get('mode') == ('normal' if route_role == 'target' else 'linklocal')
         and A.exact_json(raw.get('source'), receipt(OUTPUT / 'admission.json')['host_source']),
         'health discovery transport')
    c = raw['command']; out = base64.b64decode(c['stdout_base64'], validate=True)
    err = base64.b64decode(c['stderr_base64'], validate=True)
    need(type(c['returncode']) is int and c['returncode'] == 0 and c['timed_out'] is False
         and c['reaped'] is True and not err and len(out) <= 1024
         and A.sha(out) == c['stdout_sha256'] and A.sha(err) == c['stderr_sha256'], 'health discovery raw evidence')
    found = A.fields(out)
    need(set(found) == {'boot_id', 'bundle', 'release', 'ready'} and found['ready'] == 'ready', 'health discovery readiness')
    identity = {k: found[k] for k in ('boot_id', 'bundle', 'release')}
    excluded = [A.SOURCE_BOOT]
    if role == 'fallback' and 'observe_target' in results:
        excluded.append(results['observe_target']['identity']['boot_id'])
    C.same_identity(identity, C.TARGET if role == 'target' else C.FALLBACK, excluded=excluded)
    if name=='locate_fallback' and role=='target' and 'observe_target' in results:
        need(A.exact_json(identity,results['observe_target']['identity']),'target recovery boot changed')
    return found['boot_id']


def command_policy(name, context, results, pin, request, source):
    req = A.W.request(json.dumps(request, allow_nan=False).encode())
    need(A.exact_json(req['source'], source), 'command host source changed')
    if name in A.PHASES:
        need(req['timeout'] == 35, 'mutation command deadline')
        mode, remote = 'normal', A.SHELL
        if name in ('arm_state', 'restore_source_state', 'restore_fallback_state','restore_target_state'):
            operation = {'arm_state': 'stage', 'restore_source_state': 'restore-source', 'restore_fallback_state': 'restore-v11','restore_target_state':'restore-target'}[name]
            boot = (results['locate_fallback']['identity']['boot_id'] if name=='restore_target_state' else
                    A.SOURCE_BOOT if name!='restore_fallback_state' else results.get('observe_fallback', results['locate_fallback'])['identity']['boot_id'])
            script = A.G.build(operation, boot, A.SOURCE_BOOT, A.OWNER, A.RECEIPT_SHA)
            if name == 'restore_fallback_state': mode = 'linklocal'
        elif name=='stage_target_recovery':
            boot=results['locate_fallback']['identity']['boot_id']
            script=A.F.build_target(boot,A.SOURCE_BOOT,A.OWNER,A.RECEIPT_SHA)
        elif name == 'stage_fallback':
            boot = results.get('observe_fallback', results['locate_fallback'])['identity']['boot_id']
            script = A.F.build(boot, A.SOURCE_BOOT, A.OWNER, A.RECEIPT_SHA); mode = 'linklocal'
        else:
            action = {'install_exitrd': 'install-exitrd', 'request_source_reboot': 'request-reboot', 'restore_source_exitrd': 'restore-exitrd'}[name]
            script = A.A.build(action, A.SOURCE_BOOT, A.SOURCE_BOOT, A.OWNER, A.RECEIPT_SHA)
        expected = dict(format='rog5-one-usb-command-v1', mode=mode, remote=remote, script=script, source=source, timeout=35)
        need(A.exact_json(req, expected), 'mutation command differs from exact generator')
        expected_intent = dict(intent_sha256=pin, script_sha256=A.sha(script.encode()), mode=mode, remote=remote)
        need(A.exact_json(receipt(OUTPUT / (name + '-command-entered.json')), expected_intent), 'mutation command intent')
    else:
        need(name in H.PHASES, 'unsupported transport phase')
        role = 'target' if name in ('observe_target', 'post_capture_health','verify_target_restoration') else 'fallback'
        if name=='locate_fallback':
            if req['script'] in (H.discovery('target'),H.discovery('fallback')):
                role='target' if req['script']==H.discovery('target') else 'fallback'
            else:
                role='target' if req['remote']=='python3 -I -B -' else 'fallback'
        mode = 'normal' if role == 'target' else 'linklocal'
        if req['script'] == H.discovery(role):
            need(1 <= req['timeout'] <= 10 and req['remote'] == A.SHELL, 'discovery command scope')
            label = 'discovery'
        else:
            boot = discovered_boot(name, role, results)
            full_target=role=='target' and name!='verify_target_restoration'
            purpose='verify-restored' if name in ('verify_fallback_restoration','verify_target_restoration') else 'observe'
            script = H.H.script('target',boot) if full_target else H.F.build(purpose,boot,role)
            need(req['script']==script and req['timeout']==35
                 and req['remote']==('python3 -I -B -' if full_target else A.SHELL),'health command differs from exact generator')
            label = 'health'
        need(req['mode'] == mode, 'health command route')
        expected = dict(intent_sha256=pin, script_sha256=A.sha(req['script'].encode()), read_only=True, mode=mode)
        paths = list(OUTPUT.glob(name + ('-discovery-*-entered.json' if label == 'discovery' else '-health-entered.json')))
        need(0 < len(paths) <= (300 if name=='locate_fallback' else 150), 'health command intent absent')
        active = [p for p in paths if not p.with_name(p.name.removesuffix('-entered.json') + '-transport.json').exists()]
        need(len(active) == 1 and A.exact_json(receipt(active[0]), expected), 'health command intent binding/reuse')
    return True


def transport_authorize(name, context, results, pin, request):
    value = phase(name, context, results, pin)
    return command_policy(name, context, results, pin, request, value['host_source'])


def authorize_transport(envelope, admitted, candidate):
    need(envelope.get('format') == 'rog5-root-fallback-launch-v1', 'root transport envelope')
    ref = envelope['request']; name = ref['phase']; context = ref['context']
    need(name in ('locate_fallback', 'observe_fallback', 'stage_fallback', 'restore_fallback_state', 'verify_fallback_restoration')
         and type(ref['index']) is str and re.fullmatch('[0-9]{3}', ref['index']), 'root transport phase/index')
    raw = R.receipt(OUTPUT / (name + '-root-ssh-' + ref['index'] + '-request.json'), 4 * 1024**2)
    need(A.sha(raw) == ref['request_sha256'] and A.exact_json(A.decode(raw), candidate), 'root transport request bytes')
    need(set(candidate) == {'format', 'phase', 'context', 'intent_sha256', 'command'}
         and candidate['format'] == 'rog5-fallback-route-command-v1' and candidate['phase'] == name
         and A.exact_json(candidate['context'], context), 'root transport candidate schema')
    pin = candidate['intent_sha256']; results = results_for(context, name, pin)
    prefix = OUTPUT / (name + '-root-ssh-' + ref['index'])
    need(name not in ('stage_fallback', 'restore_fallback_state') or ref['index'] == '000',
         'root mutation retry forbidden')
    need(A.exact_json(receipt(Path(str(prefix) + '-entered.json')),
                     dict(intent_sha256=pin, request_sha256=ref['request_sha256']))
         and not os.path.lexists(Path(str(prefix) + '-process.json')), 'root transport attempt already completed or unbound')
    value = phase(name, context, results, pin, admitted, envelope)
    need(candidate['command']['mode'] == 'linklocal', 'root transport cannot use normal route')
    return command_policy(name, context, results, pin, candidate['command'], value['host_source'])


def capture_launch_authorize(req, admitted=None, envelope=None):
    role = req.get('role')
    need(role in ('target', 'fallback') and set(req) == {'format', 'role', 'context', 'source'}
         and req['format'] == 'rog5-owned-capture-request-v1', 'root capture request')
    name = 'capture_start' if role == 'target' else 'recovery_capture_start'; context = req['context']
    pin = A.sha(R.receipt(OUTPUT / (name + '-entered.json'), 16384))
    results = results_for(context, name, pin)
    value = phase(name, context, results, pin, admitted, envelope)
    need(A.exact_json(req['source'], value['host_source'])
         and A.exact_json(receipt(OUTPUT / (name + '-launch-entered.json'), 4096), req)
         and A.exact_json(receipt(OUTPUT / (name + '-command-entered.json'), 4096), dict(intent_sha256=pin, role=role)),
         'root capture source/launch binding')
    return True


def authorize_capture(envelope, admitted):
    need(envelope.get('format') == 'rog5-root-capture-launch-v1', 'root capture envelope')
    return capture_launch_authorize(envelope['request'], admitted, envelope)


def assemble(source):
    """Return the concrete driver/owned bridge only after complete qualification.

    The launcher creates the one-use Controller and writes admission.json before
    calling run(); it must close the returned bridge after terminal cleanup.
    """
    need(os.geteuid() == 1000, 'driver assembly owner')
    locked = inputs()
    need(A.exact_json(W.source_identity(), source) and source.get('clean') is True, 'assembly source')
    qualification(source)
    need(A.exact_json(claim(), B.expected_fields()), 'assembly claim')
    modules = {}
    for name in ('capture-bridge.py', 'fallback-transport.py'):
        path = STATE / 'gpu-iommu-capture-r1' / name if name == 'capture-bridge.py' else HERE / name
        row = locked['files'][str(path.relative_to(STATE))]
        digest_file(path, row['size'], row['uid'], row['mode'], row['sha256'])
        spec = importlib.util.spec_from_file_location('admitted_' + name.replace('-', '_'), path)
        module = importlib.util.module_from_spec(spec); spec.loader.exec_module(module)
        modules[name] = module
    transport_module = modules['fallback-transport.py']
    need(transport_module.OUTPUT == OUTPUT == D.OUTPUT, 'assembly transport output mismatch')
    transport = transport_module.Transport(transport_authorize)
    bridge = modules['capture-bridge.py'].Bridge(capture_launch_authorize)
    driver = D.Driver(source, before_phase, authorize, bridge, transport=transport.perform)
    return driver, bridge
