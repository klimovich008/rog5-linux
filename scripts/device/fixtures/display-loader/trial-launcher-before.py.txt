"""One fixed kernel trial launcher. Import-only; full qualification is mandatory.

No claim registration or creation. The separately prepared exact claim must
already exist; this launcher consumes it once after authentication succeeds.
"""

import fcntl
import importlib.util
import json
import os
from pathlib import Path
import stat
import time

HERE=Path(__file__).resolve().parent
LAUNCH=HERE/'trial-launch-r1'
AUTH_ROOT=HERE/'boot-authentication-r1'
LOCK=HERE/'trial-launch.lock'
ADMISSION_SHA='7c076f7812ed11b694e4c827c4bbc25fe6f7a370c860cb2c3d8ce4804b92580f'
KEEPER_SHA='629562f4de639333d474291eb5408cb615138dc735bf9ac420eb97c426df4b60'
PROBE_LAUNCHER_SHA='7265fb8db328a3bc459e745894359e45ba808926732462025408885f543deabf'


def load(name,path,digest):
    if path.resolve()!=path or path.stat().st_size>262144 or hashlib.sha256(path.read_bytes()).hexdigest()!=digest:
        raise ValueError('trial launcher dependency changed: '+path.name)
    spec=importlib.util.spec_from_file_location(name,path);m=importlib.util.module_from_spec(spec);spec.loader.exec_module(m);return m


AD=load('trial_admission',HERE/'live-admission.py',ADMISSION_SHA)
K=load('trial_credentials',HERE.parent/'oled-startup-live-driver-r1/credential-keeper.py',KEEPER_SHA)
P=load('trial_authentication',HERE.parent/'oled-startup-live-driver-r1/run-privilege-probe.py',PROBE_LAUNCHER_SHA)
G=load('trial_gpu_session',HERE.parent/'gpu-iommu-session-r1/session.py','2a2316564977a2839a31f11fb0435859857b6567f38fc5f0752ca27b5f63fa8c')
A,C,B,need=AD.A,AD.C,AD.B,AD.need


def registered_claim():
    raw=B.CLAIMS.expected_record(B.PROFILE_ID)
    need(type(raw) is bytes and len(raw)<=8192 and raw.endswith(b'\n'),'registered claim framing')
    rows=[line.split('=',1) for line in raw.decode('ascii').splitlines()]
    need(all(len(row)==2 for row in rows) and len({row[0] for row in rows})==len(rows),'registered claim fields')
    fields=dict(rows)
    need(A.exact_json(fields,B.expected_fields()),'registered claim is not this exact image/device')
    return fields



def pending_claim():
    """Verify the exact unconsumed record without creating a trial or claim."""
    claims=B.CLAIMS;expected=claims.expected_record(B.PROFILE_ID)
    root=claims.canonical_claim_root();anchor=claims.canonical_claim_anchor()
    anchor_fd,parent_fd=claims.open_claim_anchor(anchor)
    directory_fd=None
    try:
        directory_fd=claims.open_claim_root(root)
        need(not claims.existing_guard_is_exact(
            '.rog5-temporary-boot-consumption.'+B.PROFILE_ID+'.entered',anchor_fd,expected),
            'trial claim already consumed')
        name=B.PROFILE_ID+'.record'
        need(not os.path.lexists(root/(name+'.entered')),'trial claim already entered')
        claims.verify_entered_file(name,directory_fd,expected,'pending trial claim')
        claims.verify_claim_root_path(root,directory_fd)
        claims.verify_claim_anchor_path(anchor,anchor_fd,parent_fd)
        return A.sha(expected)
    finally:
        if directory_fd is not None:os.close(directory_fd)
        os.close(anchor_fd);os.close(parent_fd)


def prepared_predecessor():
    AD.PREP.prior_and_staging()


def prepare():
    """Read-only preparation. Missing qualification refuses before any attempt."""
    need(os.geteuid()==1000 and LAUNCH.resolve()==LAUNCH,'trial owner/path')
    need(not os.path.lexists(LAUNCH) and not os.path.lexists(AD.OUTPUT),'trial was already attempted')
    for flag in ('ALLOW_TEMPORARY_BOOT','ALLOW_HEADLESS_LIVE_GATE'):
        need(os.environ.get(flag)=='1','explicit trial execution environment required')
    prepared_predecessor()
    AD.inputs();G.cohort(AD);source=AD.W.source_identity()
    need(source.get('clean') is True,'trial source is dirty')
    qualification=AD.qualification(source)
    fields=registered_claim();pending_claim()
    return dict(source=source,qualification_sha256=qualification,claim_fields=fields,
                launcher_sha256=A.sha(Path(__file__).read_bytes()))


def record(prepared,expires):
    identity=AD.S.process_identity(os.getpid())
    need(identity['uids']==[1000]*4,'trial controller identity')
    now=time.monotonic()
    need(now<expires<=now+AD.MAX_LIFETIME,'trial admission lifetime')
    return dict(format='rog5-kernel-driver-admission-v1',status='ADMITTED',
        host_source=prepared['source'],source_phone=dict(C.SOURCE,boot_id=A.SOURCE_BOOT),owner=A.OWNER,
        host_boot_id=Path('/proc/sys/kernel/random/boot_id').read_text().strip(),
        controller=dict(pid=os.getpid(),start=identity['start'],uid=1000),inputs_sha256=AD.INPUTS_SHA,
        root_capture_verifier_sha256=ADMISSION_SHA,root_transport_verifier_sha256=ADMISSION_SHA,
        launcher_sha256=prepared['launcher_sha256'],qualification_sha256=prepared['qualification_sha256'],
        claim_fields=prepared['claim_fields'],created_monotonic=now,expires_monotonic=expires)


class Driver:
    PHASES=C.PHASES
    def __init__(self,inner,credentials):self.inner,self.credentials=inner,credentials
    def gate(self,*args):
        self.credentials.check()
        return self.inner.gate(*args)
    def __getattr__(self,name):
        if name not in self.PHASES:raise AttributeError(name)
        return getattr(self.inner,name)


def close_owned(driver,bridge):
    errors=[]
    if driver is not None:
        for worker in driver.captures.workers.values():
            if not worker.closed:
                errors.append('unfinished owned recording required cancellation')
                try:worker.stop();worker.finish()
                except Exception as error:errors.append(str(error)[:1000])
    if bridge is not None:
        try:bridge.close()
        except Exception as error:errors.append(str(error)[:1000])
    return errors


def _run_once(*,authenticate=False):
    need(type(authenticate) is bool,'trial authentication mode')
    prepared=prepare()  # Nothing below runs while any preparation gate is absent.
    LAUNCH.mkdir(mode=0o700)
    C.save(LAUNCH/'entered.json',dict(prepared=prepared,controller_pid=os.getpid(),
        authenticate=authenticate,phone_actions_started=False))
    inner=None;bridge=None;trial=None;gpu=None;error=None;cleanup=[];claim_entered=False
    try:
        if authenticate:P.authenticate(LAUNCH)
        with K.Credentials(LAUNCH/'credentials') as credentials:
            AD.inputs()
            need(A.exact_json(AD.W.source_identity(),prepared['source'])
                 and AD.qualification(prepared['source'])==prepared['qualification_sha256'],
                 'trial inputs changed during authentication')
            need(A.exact_json(registered_claim(),prepared['claim_fields']),'trial registry changed')
            pending_claim()
            preparation_pin=AD.PREP.observe_fresh(prepared['source'])
            C.save(LAUNCH/'claim-consumption-entered.json',dict(profile=B.PROFILE_ID,attempt_limit=1))
            claim_entered=True
            B.CLAIMS.consume(B.PROFILE_ID)
            B.canonical(AD.D.BOOT.IMAGE_SHA,B.SERIAL)
            C.save(LAUNCH/'claim-consumption-result.json',dict(status='PASS',profile=B.PROFILE_ID))
            inner,bridge=AD.assemble(prepared['source'])
            admitted=record(prepared,credentials.started+AD.MAX_LIFETIME)
            admitted['source_preparation_sha256']=preparation_pin
            AD.PREP.verify_ready(preparation_pin,prepared['source'],admitted['created_monotonic'])
            raw=(json.dumps(admitted,sort_keys=True,indent=2,allow_nan=False)+'\n').encode()
            context=dict(source=admitted['source_phone'],owner=A.OWNER,admission_sha256=A.sha(raw))
            controller=C.Controller(Driver(inner,credentials),context)
            need(C.save(AD.OUTPUT/'admission.json',admitted)==context['admission_sha256'],'trial admission encoding')
            try:
                trial=controller.run()
                if trial.get('status')=='COMPONENT_PASS':
                    gpu=G.run(AD,controller,credentials,trial)
                credentials.check()
            finally:
                cleanup=close_owned(inner,bridge)
                inner=None;bridge=None
    except Exception as exc:error=dict(type=type(exc).__name__,reason=str(exc)[:1000])
    finally:
        if inner is not None or bridge is not None:cleanup.extend(close_owned(inner,bridge))
    result=dict(status='COMPONENT_PASS' if trial and trial.get('status')=='COMPONENT_PASS' and gpu and gpu.get('status')=='PASS_GPU_INITIALIZATION_SESSION' and error is None and not cleanup else 'FAIL',
        trial=trial,gpu_session=gpu,error=error,cleanup_errors=cleanup,claim_consumption_entered=claim_entered,
        retry_permitted=False,release_qualified=False)
    C.save(LAUNCH/'result.json',result)
    return result


def closed_authentication_attempts():
    """A released flock alone does not prove an interrupted sudo child exited."""
    if not os.path.lexists(AUTH_ROOT):
        return
    st=AUTH_ROOT.lstat()
    need(AUTH_ROOT.resolve()==AUTH_ROOT and stat.S_ISDIR(st.st_mode)
         and st.st_uid==1000 and stat.S_IMODE(st.st_mode)==0o700,'authentication root metadata')
    for attempt in AUTH_ROOT.iterdir():
        st=attempt.lstat()
        need(attempt.resolve()==attempt and stat.S_ISDIR(st.st_mode) and st.st_uid==1000
             and stat.S_IMODE(st.st_mode)==0o700,'authentication attempt metadata')
        owner=A.decode(AD.R.receipt(attempt/'owner.json',8192))
        result=A.decode(AD.R.receipt(attempt/'result.json',8192))
        auth=A.decode(AD.R.receipt(attempt/'authentication-result.json',8192))
        need(result.get('status')=='FAIL_AUTHENTICATION' and result.get('trial') is None
             and result.get('claim_consumption_entered') is False and result.get('cleanup_errors')==[],
             'prior authentication outcome is not a closed failure')
        need(auth.get('reaped') is True and type(auth.get('returncode')) is int
             and auth.get('password_retained') is False,'prior authentication child is not reaped')
        controller=owner['controller']
        need(type(controller.get('pid')) is int and controller['pid']>1
             and type(controller.get('start')) is str and controller['start'].isdigit(),
             'prior authentication controller identity')
        if owner.get('host_boot_id')==Path('/proc/sys/kernel/random/boot_id').read_text().strip():
            try:
                current=AD.S.process_identity(controller['pid'])
            except FileNotFoundError:
                pass
            else:
                need(current['start']!=controller['start'],'prior authentication controller remains')


def run(*,authenticate=False):
    """Keep password attempts separate from the exclusive one-use boot entry."""
    need(type(authenticate) is bool,'trial authentication mode')
    prepare()  # Read-only refusal before lock, directory or password window.
    fd=os.open(LOCK,os.O_RDWR|os.O_CREAT|os.O_NOFOLLOW|os.O_CLOEXEC,0o600)
    try:
        st=os.fstat(fd)
        need(LOCK.resolve()==LOCK and stat.S_ISREG(st.st_mode) and st.st_uid==1000
             and st.st_nlink==1 and stat.S_IMODE(st.st_mode)==0o600,'trial lock metadata')
        fcntl.flock(fd,fcntl.LOCK_EX|fcntl.LOCK_NB)
        need(LOCK.stat().st_ino==st.st_ino and LOCK.stat().st_dev==st.st_dev,'trial lock changed')
        prepare()  # Recheck after winning exclusive ownership.
        closed_authentication_attempts()
        if authenticate:
            AUTH_ROOT.mkdir(mode=0o700,exist_ok=True)
            st=AUTH_ROOT.lstat()
            need(AUTH_ROOT.resolve()==AUTH_ROOT and stat.S_ISDIR(st.st_mode)
                 and st.st_uid==1000 and stat.S_IMODE(st.st_mode)==0o700,'authentication root metadata')
            attempt=AUTH_ROOT/('attempt-'+os.urandom(16).hex())
            attempt.mkdir(mode=0o700)
            identity=AD.S.process_identity(os.getpid())
            C.save(attempt/'owner.json',dict(controller=dict(pid=os.getpid(),start=identity['start']),
                host_boot_id=Path('/proc/sys/kernel/random/boot_id').read_text().strip(),
                launcher_sha256=A.sha(Path(__file__).read_bytes()),phone_actions=0,claim_actions=0))
            try:
                P.authenticate(attempt)
            except Exception as exc:
                result=dict(status='FAIL_AUTHENTICATION',trial=None,
                    error=dict(type=type(exc).__name__,reason=str(exc)[:1000]),cleanup_errors=[],
                    claim_consumption_entered=False,retry_permitted=False,release_qualified=False)
                C.save(attempt/'result.json',result)
                return result
            C.save(attempt/'result.json',dict(status='PASS_AUTHENTICATION',phone_actions=0,claim_actions=0))
        # Credentials still belong to this process/session. No boot entry exists
        # until authentication has returned successfully. All old boot guards,
        # credential checks, claim consumption and cleanup remain in run_once.
        return _run_once(authenticate=False)
    finally:
        os.close(fd)  # Retain the lock inode; never unlink a shared flock file.
